I

Interface Segregation — สร้าง Interface เล็กๆ ก่อน

เริ่มจาก Interface เพราะทุกอย่างในระบบจะ depend กัน ต้องวางรากฐานให้ดีก่อน

💡 แทนที่จะทำ Interface ใหญ่ตัวเดียว เราแยกเป็น Interface เล็กๆ ตามหน้าที่ — ทั้ง SendMessage handler, UnityEvents handler, หรือ AI Bot ล้วน implement IInputHandler เดียวกัน PlayerController จึงไม่รู้ว่า input มาจากไหน
Input / IInputHandler.cs 
// [I] Interface Segregation + [D] Dependency Inversion
// Interface นี้ไม่สนใจว่า input มาจาก PlayerInput SendMessage,
// UnityEvents, legacy Input หรือ AI
// PlayerController รู้จักแค่ตัวนี้เท่านั้น
public interface IInputHandler
{
    // ทิศทางเคลื่อนที่ (ค่า -1 ถึง 1 ทั้ง x และ y)
    Vector2 GetMoveInput();

    // true เฉพาะเฟรมที่กดปุ่มยิง
    bool GetFireInput();
}
Combat / IDamageable.cs 
// [I] Interface Segregation
// แยก "รับ damage" ออกเป็น Interface ของตัวเอง
// Player, Enemy, หรือแม้แต่ Barrel ที่ระเบิดได้ ล้วน implement ได้
// โดยไม่ต้องมี inheritance เดียวกัน
public interface IDamageable
{
    void TakeDamage(int amount);
}
D

Dependency Inversion — PlayerInput Component ผ่าน IInputHandler

PlayerController ไม่รู้จัก PlayerInput โดยตรง — รู้จักแค่ IInputHandler

🎮 ทำไมต้อง wrap PlayerInput ด้วย Interface? — ถ้า PlayerController เรียก PlayerInput โดยตรง จะทดสอบไม่ได้ เพิ่ม AI bot ไม่ได้ และเปลี่ยน input method ต้องแก้โค้ด แต่ถ้า wrap ไว้ใน IInputHandler แค่สลับ component ใน Inspector ก็เปลี่ยนได้ทันที
📋
ตั้งค่า PlayerInput Component ใน Unity Editor

1. Package Manager → ค้นหา Input System → Install → Restart
2. Project Settings → Player → Active Input Handling → Input System Package (New)
3. เลือก Player GameObject → Add Component → Player Input
4. คลิก Create Actions → บันทึกเป็น PlayerInputActions
5. ใน Actions Asset: Action Map "Player" มี Action Move (Vector2) และ Fire (Button)
6. Behavior → เลือก "Send Messages" หรือ "Invoke Unity Events"
7. เพิ่ม Component PlayerInputHandler (script ด้านล่าง) บน GameObject เดียวกัน
Input / PlayerInputHandler.cs  —  Behavior: Send Messages 
using UnityEngine;
using UnityEngine.InputSystem;

// [D] Concrete implementation ของ IInputHandler โดยใช้ PlayerInput + SendMessage
// PlayerInput Component จะเรียก OnMove / OnFire ผ่าน SendMessage อัตโนมัติ
// เมื่อผู้เล่นกด input ที่ตรงกับ Action ใน Action Asset
//
// ⚠️ SendMessage ทำงานได้เฉพาะเมื่อ script นี้อยู่บน GameObject
//    เดียวกับ PlayerInput Component เท่านั้น
//
// [I] implement เฉพาะ method ที่ต้องการ ไม่มีอะไรเกิน
// [O] ถ้าอยากรองรับ AI bot แค่สร้าง AIInputHandler ใหม่ class เดียว
public class PlayerInputHandler : MonoBehaviour, IInputHandler
{
    // PlayerInput Component จะ inject ค่าเหล่านี้ผ่าน SendMessage callbacks
    // script นี้ไม่ต้อง GetComponent หรือ reference PlayerInput เอง
    private Vector2 moveInput;
    private bool firePressedThisFrame;

    // ----- SendMessage Callbacks -----
    // ชื่อ method ต้องตรงกับ "On" + ชื่อ Action ใน Action Asset เป๊ะ
    // เช่น Action "Move" → OnMove, Action "Fire" → OnFire

    // called ทุกครั้งที่ค่า Move เปลี่ยน (กด/ปล่อย WASD หรือขยับ Stick)
    void OnMove(InputValue value)
    {
        moveInput = value.Get<Vector2>();
    }

    // called เมื่อกดปุ่ม Fire (started phase เท่านั้น ไม่ triggered ซ้ำตอนค้างกด)
    void OnFire(InputValue value)
    {
        // isPressed ตรวจว่ากดอยู่หรือเปล่า — set flag เพื่อให้ PlayerController อ่านได้
        if (value.isPressed)
            firePressedThisFrame = true;
    }

    // ----- IInputHandler implementation -----

    public Vector2 GetMoveInput() => moveInput;

    public bool GetFireInput()
    {
        // อ่านค่าแล้ว clear ทันที — ป้องกันยิงซ้ำถ้า Update() เร็วกว่า callback
        if (firePressedThisFrame)
        {
            firePressedThisFrame = false;
            return true;
        }
        return false;
    }
}
Input / PlayerInputHandlerEvents.cs  —  Behavior: Invoke Unity Events 
using UnityEngine;
using UnityEngine.InputSystem;

// [D] แบบที่ 2: ใช้ Invoke Unity Events แทน SendMessage
// ต่างกันที่: method signature ต้องรับ InputAction.CallbackContext
// และต้อง wire ใน Inspector ของ PlayerInput Component เอง
//
// ข้อดีของ UnityEvents vs SendMessage:
//   UnityEvents → เห็น connection ชัดใน Inspector, type-safe กว่า, debug ง่าย
//   SendMessage → setup น้อยกว่า แต่ connection ซ่อนอยู่ใน code
public class PlayerInputHandlerEvents : MonoBehaviour, IInputHandler
{
    private Vector2 moveInput;
    private bool firePressedThisFrame;

    // ----- UnityEvent target methods -----
    // ใน Inspector ของ PlayerInput Component:
    //   Events → Player → Move → ลาก GameObject แล้วเลือก method นี้
    //   Events → Player → Fire → ลาก GameObject แล้วเลือก method นี้
    //
    // Method signature ต้องเป็น (InputAction.CallbackContext) เสมอ

    // wired ไปที่ Events → Player → Move (performed)
    public void OnMove(InputAction.CallbackContext context)
    {
        // performed = มีค่า, canceled = ปล่อยปุ่ม (ค่าเป็น zero)
        moveInput = context.ReadValue<Vector2>();
    }

    // wired ไปที่ Events → Player → Fire (started) — triggered ครั้งเดียวตอนกด
    public void OnFire(InputAction.CallbackContext context)
    {
        // started phase = กดครั้งแรก เหมาะสำหรับ "ยิง 1 นัดต่อการกด"
        if (context.phase == InputActionPhase.Started)
            firePressedThisFrame = true;
    }

    // ----- IInputHandler implementation -----

    public Vector2 GetMoveInput() => moveInput;

    public bool GetFireInput()
    {
        if (firePressedThisFrame)
        {
            firePressedThisFrame = false;
            return true;
        }
        return false;
    }
}
⚖️ เปรียบเทียบ SendMessage vs Invoke Unity Events 
// ============================================================
//  Feature           | Send Messages       | Invoke Unity Events
// ============================================================
//  Setup             | เพิ่ม script อย่าง  | ต้อง wire ใน Inspector
//                    | เดียว               | ทีละ Action ทีละ method
// ------------------------------------------------------------
//  Method signature  | OnMove(InputValue)  | OnMove(CallbackContext)
// ------------------------------------------------------------
//  Connection        | ซ่อนอยู่ใน naming  | เห็นใน Inspector ชัดเจน
//  visibility        | convention          |
// ------------------------------------------------------------
//  Type safety       | ต่ำกว่า             | สูงกว่า
// ------------------------------------------------------------
//  Performance       | SendMessage ช้ากว่า | เร็วกว่า (direct invoke)
//                    | (reflection)        |
// ------------------------------------------------------------
//  Multi-component   | broadcast ได้ทุก   | เลือก wire เฉพาะที่ต้องการ
//                    | component บน GO    |
// ------------------------------------------------------------
//  ข้อสรุป           | เหมาะสำหรับ        | เหมาะสำหรับ production
//                    | prototype เร็ว     | และ team ใหญ่
// ============================================================
//
//  ทั้งสองแบบ implement IInputHandler เหมือนกัน
//  → PlayerController ไม่รู้และไม่สนว่าใช้แบบไหน (D)
✨ SOLID ในทางปฏิบัติ: drag PlayerInputHandler หรือ PlayerInputHandlerEvents ใส่ Inspector ของ PlayerController ก็ใช้ได้ทั้งคู่ — หรือจะสร้าง AIInputHandler ใหม่ก็ไม่ต้องแก้ PlayerController แม้แต่บรรทัดเดียว นี่คือ D + I + O ทำงานร่วมกัน
S

Single Responsibility — 1 Class = 1 หน้าที่

แทนที่ Player class เดียวจะทำทุกอย่าง แยกออกเป็น Controller, Mover, Health

✂️ PlayerController รับ input แล้วสั่งการ  |  PlayerMover คำนวณ movement  |  PlayerHealth จัดการ HP — แก้ไข movement ไม่กระทบ HP logic และกลับกัน
Player / PlayerController.cs 
using UnityEngine;

// [S] หน้าที่เดียว: รับ Input แล้วสั่งให้ component อื่นทำงาน
// [D] พึ่ง IInputHandler Interface — ไม่รู้จัก NewInputHandler หรือ AIInputHandler โดยตรง
// เปลี่ยน input source ทำได้ใน Inspector โดยไม่แก้โค้ดนี้เลย
public class PlayerController : MonoBehaviour
{
    // drag NewInputHandler หรือ AIInputHandler ใส่ใน Inspector ก็ได้
    // PlayerController ไม่สนว่าเป็น concrete class อะไร
    [SerializeField] private MonoBehaviour inputHandlerMono;
    [SerializeField] private PlayerMover mover;
    [SerializeField] private Weapon currentWeapon;

    private IInputHandler input; // เก็บเป็น Interface ไม่ใช่ concrete class

    void Start()
    {
        // Cast MonoBehaviour → Interface ตอน runtime
        input = inputHandlerMono as IInputHandler;

        if (input == null)
            Debug.LogError("inputHandlerMono ต้อง implement IInputHandler!");
    }

    void Update()
    {
        // สั่ง Mover ให้เคลื่อนที่ — ไม่คำนวณ movement เอง (S)
        // GetMoveInput() อาจมาจาก WASD, Gamepad Stick, หรือ AI — ไม่สนใจ (D)
        mover.Move(input.GetMoveInput());

        // สั่ง Weapon ให้ยิง — ไม่รู้ว่า weapon ประเภทไหน (O)
        if (input.GetFireInput())
            currentWeapon.Fire();
    }
}
Player / PlayerMover.cs 
using UnityEngine;

// [S] หน้าที่เดียว: คำนวณและ apply การเคลื่อนที่
// ไม่รู้เรื่อง Input, HP, หรือ Score ทั้งสิ้น
public class PlayerMover : MonoBehaviour
{
    [SerializeField] private float speed = 5f;

    public void Move(Vector2 direction)
    {
        // normalize เพื่อไม่ให้วิ่งเร็วขึ้นตอนกด diagonal
        Vector2 normalized = direction.normalized;
        transform.Translate(
            new Vector3(normalized.x, normalized.y, 0)
            * speed * Time.deltaTime
        );
    }
}
Player / PlayerHealth.cs 
using UnityEngine;

// [S] หน้าที่เดียว: จัดการ HP ของ Player
// [I] implement IDamageable — ใครก็ยิงหา Player ได้โดยไม่รู้จัก class นี้
// ใช้ Action event เพื่อแจ้ง UIManager โดยไม่ต้อง reference กันตรงๆ
public class PlayerHealth : MonoBehaviour, IDamageable
{
    [SerializeField] private int maxHP = 100;
    private int currentHP;

    // Events — UIManager subscribe ตรงนี้ PlayerHealth ไม่ต้องรู้จัก UI
    public System.Action<int> OnHealthChanged;
    public System.Action OnPlayerDied;

    void Start() => currentHP = maxHP;

    public void TakeDamage(int amount)
    {
        currentHP = Mathf.Max(0, currentHP - amount); // clamp ไม่ให้ติดลบ
        OnHealthChanged?.Invoke(currentHP);              // แจ้ง UI

        if (currentHP <= 0)
            OnPlayerDied?.Invoke();
    }

    public int GetCurrentHP() => currentHP;
    public int GetMaxHP() => maxHP;
}
O

Open/Closed — เพิ่ม Weapon ใหม่ โดยไม่แก้ของเดิม

สร้าง abstract Weapon แล้วแต่ละ weapon เป็น subclass — เพิ่มได้เรื่อยๆ ไม่มีวันสิ้นสุด

🔫 PlayerController เรียกแค่ currentWeapon.Fire() — ไม่รู้ว่าตอนนี้ถืออะไร จะเพิ่ม Sniper, Flamethrower, หรือ Laser Cannon ก็ไม่ต้องแก้ PlayerController แม้แต่บรรทัดเดียว
Combat / Weapon.cs 
using UnityEngine;

// [O] Abstract base class — "เปิด" ให้ต่อขยาย แต่ "ปิด" ไม่ให้แก้ตรงนี้
// PlayerController รู้จักแค่ class นี้ — ไม่รู้จัก Pistol, Shotgun โดยตรง
public abstract class Weapon : MonoBehaviour
{
    [SerializeField] protected GameObject bulletPrefab;
    [SerializeField] protected Transform firePoint;   // จุดที่กระสุนออก
    [SerializeField] protected float fireRate = 0.5f;  // ดีเลย์ระหว่างช็อต

    protected float nextFireTime;

    // subclass ต้อง override — นี่คือจุดที่ "ต่อขยาย" เกิดขึ้น
    public abstract void Fire();

    // helper method ให้ subclass ใช้ตรวจสอบ cooldown
    protected bool CanFire()
    {
        return Time.time >= nextFireTime;
    }
}
Combat / Pistol.cs 
using UnityEngine;

// [O] ต่อขยาย Weapon โดยไม่แก้ base class
public class Pistol : Weapon
{
    public override void Fire()
    {
        if (!CanFire()) return;

        // ยิงกระสุน 1 นัดตรงๆ
        Instantiate(bulletPrefab, firePoint.position, firePoint.rotation);
        nextFireTime = Time.time + fireRate;
    }
}
Combat / Shotgun.cs 
using UnityEngine;

// [O] เพิ่ม Weapon ใหม่ — Pistol และ PlayerController ไม่ถูกแตะต้องเลย
public class Shotgun : Weapon
{
    [SerializeField] private int pelletCount = 5;
    [SerializeField] private float spreadAngle = 20f;

    public override void Fire()
    {
        if (!CanFire()) return;

        // ยิง pellet กระจาย spread angle
        for (int i = 0; i < pelletCount; i++)
        {
            float angle = Random.Range(-spreadAngle / 2, spreadAngle / 2);
            Quaternion rot = firePoint.rotation * Quaternion.Euler(0, 0, angle);
            Instantiate(bulletPrefab, firePoint.position, rot);
        }

        nextFireTime = Time.time + fireRate;
    }
}
Combat / RocketLauncher.cs 
using UnityEngine;

// [O] เพิ่ม weapon ที่ 3 — ขยายระบบโดยสมบูรณ์ ไม่มีการแก้ไขใดๆ ก่อนหน้า
public class RocketLauncher : Weapon
{
    [SerializeField] private float explosionRadius = 3f;

    public override void Fire()
    {
        if (!CanFire()) return;

        GameObject rocket = Instantiate(
            bulletPrefab, firePoint.position, firePoint.rotation
        );

        // ตั้ง explosion radius ให้ rocket component (ถ้ามี)
        Rocket rocketComp = rocket.GetComponent<Rocket>();
        if (rocketComp != null)
            rocketComp.SetExplosionRadius(explosionRadius);

        nextFireTime = Time.time + fireRate;
    }
}
L

Liskov Substitution — Enemy ทุกตัวใช้แทนกันได้

ทุกที่ที่รับ EnemyBase สามารถรับ WalkEnemy, FlyEnemy, ShooterEnemy ได้โดยไม่พัง

🔄 EnemyFactory คืน EnemyBase — ผู้เรียกไม่รู้และไม่จำเป็นต้องรู้ว่าได้ enemy ประเภทไหน พฤติกรรม TakeDamage และ Die ทำงานถูกต้องเหมือนกันในทุก subclass
Enemy / EnemyBase.cs 
using UnityEngine;

// [L] Base class ที่ subclass ทุกตัวต้องใช้แทนได้อย่างถูกต้อง
// [I] implement IDamageable — bullet ไม่ต้องรู้ว่าชน enemy ประเภทไหน
public abstract class EnemyBase : MonoBehaviour, IDamageable
{
    [SerializeField] protected int hp = 30;
    [SerializeField] protected int damage = 10;
    [SerializeField] protected int scoreValue = 100;

    protected Transform player;

    protected virtual void Start()
    {
        GameObject playerObj = GameObject.FindGameObjectWithTag("Player");
        if (playerObj != null) player = playerObj.transform;
    }

    // [L] TakeDamage ทำงานเหมือนกันทุก subclass — ไม่มีการ throw exception
    // หรือทำพฤติกรรมที่ขัดแย้งกับ base class
    public virtual void TakeDamage(int amount)
    {
        hp -= amount;
        if (hp <= 0) Die();
    }

    // subclass override ได้ถ้าอยาก die effect พิเศษ
    protected virtual void Die()
    {
        ScoreManager.Instance?.AddScore(scoreValue);
        Destroy(gameObject);
    }
}
Enemy / WalkEnemy.cs 
using UnityEngine;

// [L] ใช้แทน EnemyBase ได้ — TakeDamage ยัง work, Die ยัง work
public class WalkEnemy : EnemyBase
{
    [SerializeField] private float speed = 2f;

    void Update()
    {
        if (player == null) return;

        // เดินตรงหาผู้เล่น
        transform.position = Vector2.MoveTowards(
            transform.position,
            player.position,
            speed * Time.deltaTime
        );
    }

    // ชนผู้เล่นแล้วทำ damage
    void OnTriggerEnter2D(Collider2D col)
    {
        if (!col.CompareTag("Player")) return;

        col.GetComponent<IDamageable>()?.TakeDamage(damage);
    }
}
Enemy / FlyEnemy.cs 
using UnityEngine;

// [L] FlyEnemy ใช้แทน EnemyBase ได้ — บินเร็วกว่าแต่พฤติกรรมหลักเหมือนกัน
public class FlyEnemy : EnemyBase
{
    [SerializeField] private float speed = 4f;     // เร็วกว่า WalkEnemy
    [SerializeField] private float zigzagAmount = 2f; // บินแบบ zigzag

    void Update()
    {
        if (player == null) return;

        Vector2 dir = (player.position - transform.position).normalized;

        // เพิ่ม zigzag movement ให้หลบกระสุนยากขึ้น
        Vector2 zigzag = Vector2.Perpendicular(dir)
            * Mathf.Sin(Time.time * 3f) * zigzagAmount;

        transform.Translate((dir + zigzag * 0.3f) * speed * Time.deltaTime);
    }

    void OnTriggerEnter2D(Collider2D col)
    {
        if (!col.CompareTag("Player")) return;
        col.GetComponent<IDamageable>()?.TakeDamage(damage);
    }
}
Enemy / ShooterEnemy.cs 
using UnityEngine;

// [L] ShooterEnemy ยิงกระสุนแทนการพุ่งชน — แต่ยังใช้แทน EnemyBase ได้
// TakeDamage และ Die ยังทำงานถูกต้องเหมือนเดิม (ไม่ throw exception)
public class ShooterEnemy : EnemyBase
{
    [SerializeField] private GameObject bulletPrefab;
    [SerializeField] private float fireRate = 2f;
    [SerializeField] private float bulletSpeed = 5f;
    [SerializeField] private float attackRange = 8f;

    private float nextFireTime;

    void Update()
    {
        if (player == null) return;

        float dist = Vector2.Distance(transform.position, player.position);

        // ยิงเมื่ออยู่ในระยะและ cooldown หมดแล้ว
        if (dist <= attackRange && Time.time >= nextFireTime)
        {
            Shoot();
            nextFireTime = Time.time + fireRate;
        }
    }

    private void Shoot()
    {
        if (bulletPrefab == null) return;

        Vector2 dir = (player.position - transform.position).normalized;
        GameObject bullet = Instantiate(bulletPrefab, transform.position, Quaternion.identity);

        Rigidbody2D rb = bullet.GetComponent<Rigidbody2D>();
        if (rb != null)
            rb.linearVelocity = dir * bulletSpeed;
    }
}

using UnityEngine;

// [O] เพิ่ม EnemyType ใหม่ได้โดยเพิ่ม case ใน switch — Factory logic ไม่เปลี่ยน
// [L] คืน EnemyBase — ผู้เรียกใช้ได้โดยไม่ต้องรู้ว่าเป็น subclass อะไร
public enum EnemyType { Walk, Fly, Shooter }

public class EnemyFactory : MonoBehaviour
{
    [SerializeField] private GameObject walkEnemyPrefab;
    [SerializeField] private GameObject flyEnemyPrefab;
    [SerializeField] private GameObject shooterEnemyPrefab;

    // คืน EnemyBase — ผู้เรียกไม่ต้องรู้ type จริงๆ (L)
    public EnemyBase Spawn(EnemyType type, Vector3 position)
    {
        GameObject prefab = type switch
        {
            EnemyType.Walk    => walkEnemyPrefab,
            EnemyType.Fly     => flyEnemyPrefab,
            EnemyType.Shooter => shooterEnemyPrefab,
            _ => throw new System.ArgumentException($"Unknown EnemyType: {type}")
        };

        return Instantiate(prefab, position, Quaternion.identity)
            .GetComponent<EnemyBase>();
    }
}

Score & UI — แต่ละ class มีหน้าที่ชัดเจน

ScoreManager ไม่รู้จัก UI, UIManager ไม่ compute logic — ต่อกันผ่าน Events

Score / ScoreManager.cs 
using UnityEngine;

// [S] หน้าที่เดียว: เก็บและเพิ่ม Score
// ไม่รู้จัก UI เลย — แจ้งผ่าน event Action
public class ScoreManager : MonoBehaviour
{
    public static ScoreManager Instance; // Singleton
    public System.Action<int> OnScoreChanged;

    private int score = 0;

    void Awake()
    {
        if (Instance == null) Instance = this;
        else Destroy(gameObject);
    }

    public void AddScore(int amount)
    {
        score += amount;
        OnScoreChanged?.Invoke(score); // แจ้ง subscriber ทุกคน
    }

    public int GetScore() => score;
}
UI / UIManager.cs 
using UnityEngine;
using TMPro;

// [S] หน้าที่เดียว: แสดงผล UI
// ไม่มี game logic ใดๆ — subscribe event แล้ว update text เท่านั้น
public class UIManager : MonoBehaviour
{
    [SerializeField] private TextMeshProUGUI hpText;
    [SerializeField] private TextMeshProUGUI scoreText;
    [SerializeField] private GameObject gameOverPanel;
    [SerializeField] private PlayerHealth playerHealth;

    void Start()
    {
        // Subscribe events — UIManager ไม่ต้องรู้ว่า HP คำนวณยังไง
        playerHealth.OnHealthChanged += hp => {
            hpText.text = $"HP: {hp}";
        };

        playerHealth.OnPlayerDied += () => {
            hpText.text = "HP: 0";
            if (gameOverPanel != null)
                gameOverPanel.SetActive(true);
        };

        // Subscribe score event — ไม่ต้องรู้จัก EnemyBase หรือ logic การให้คะแนน
        ScoreManager.Instance.OnScoreChanged += score => {
            scoreText.text = $"Score: {score}";
        };

        // initialize ค่าเริ่มต้น
        hpText.text = $"HP: {playerHealth.GetMaxHP()}";
        scoreText.text = "Score: 0";
    }
}
